  <!-- Keyboard Detection Script -->
  <script>
    document.addEventListener("keydown", (e) => {
      document.getElementById("keyName").innerText =
        e.key === " " ? "Space" : e.key;

      document.getElementById("keyCode").innerText = e.keyCode;

      let mods = [];
      if (e.ctrlKey) mods.push("Ctrl");
      if (e.shiftKey) mods.push("Shift");
      if (e.altKey) mods.push("Alt");

      document.getElementById("modifiers").innerText =
        mods.length ? mods.join(" + ") : "None";
    });
  </script>
